-- Default Keybinds Table
local Keybinds = {
    PlayerCam = Enum.KeyCode.P,
    NpcCam = Enum.KeyCode.C,
    SuperJump = Enum.KeyCode.L,
    LongDash = Enum.KeyCode.H,
    TpEnemy = Enum.KeyCode.T,
    TpAlly = Enum.KeyCode.U,
    MenuToggle = Enum.KeyCode.N
}

-- Save keybinds to file function
local function saveKeybinds()
    if writefile and HttpService then
        local encodedTable = {}
        for keyName, keyCode in pairs(Keybinds) do
            encodedTable[keyName] = keyCode.Name
        end
        pcall(function()
            writefile(saveFileName, HttpService:JSONEncode(encodedTable))
        end)
    end
end

-- Load saved keybinds if file exists
local function loadKeybinds()
    if writefile and readfile and isfile and isfile(saveFileName) then
        local success, decoded = pcall(function()
            return HttpService:JSONDecode(readfile(saveFileName))
        end)
        if success and type(decoded) == "table" then
            for keyName, enumName in pairs(decoded) do
                if Keybinds[keyName] ~= nil and Enum.KeyCode[enumName] then
                    Keybinds[keyName] = Enum.KeyCode[enumName]
                end
            end
        end
    end
end

-- Variables & Logic
local playerCamlockActive, npcCamlockActive, currentTarget = false, false, nil
local enemiesFolder = Workspace:FindFirstChild("Enemies") or Workspace

local function isPlayerAlly(player)
    if player == localPlayer then return true end
    local success, isAlly = pcall(function()
        return player:GetAttribute("CrewMember") == true or player:GetAttribute("Ally") == true
    end)
    if success and isAlly then return true end
    if localPlayer.Team and player.Team and localPlayer.Team == player.Team then
        if localPlayer.Team.Name:lower() ~= "pirates" then return true end
    end
    return false
end

local function getClosestPlayer()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and not isPlayerAlly(p) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (char.HumanoidRootPart.Position - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortest then shortest, closest = dist, p.Character end
            end
        end
    end
    return closest
end

local function getClosestAlly()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and isPlayerAlly(p) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (char.HumanoidRootPart.Position - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortest then shortest, closest = dist, p.Character end
            end
        end
    end
    return closest
end

local function getClosestNPC()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    for _, obj in ipairs(enemiesFolder:GetChildren()) do
        if obj:IsA("Model") and obj ~= char then
            local hum, root = obj:FindFirstChildOfClass("Humanoid"), obj:FindFirstChild("HumanoidRootPart")
            if hum and root and not Players:GetPlayerFromCharacter(obj) and hum.Health > 0 then
                local dist = (char.HumanoidRootPart.Position - root.Position).Magnitude
                if dist < shortest then shortest, closest = dist, obj end
            end
        end
    end
    return closest
end

local function updateStatusText()
    if playerCamlockActive then StatusLabel.Text, StatusLabel.TextColor3 = "Cam: Player ON", Color3.fromRGB(0, 255, 128)
    elseif npcCamlockActive then StatusLabel.Text, StatusLabel.TextColor3 = "Cam: NPC ON", Color3.fromRGB(0, 255, 128)
    else StatusLabel.Text, StatusLabel.TextColor3 = "Cam: OFF", Color3.fromRGB(255, 50, 50) end
end

-- Helper to accurately extract Blox Fruits Level
local function getPlayerLevel(player)
    local lvl = player:GetAttribute("Level")
    if not lvl and player:FindFirstChild("Data") then
        local data = player.Data
        if data:FindFirstChild("Level") then lvl = data.Level.Value end
    end
    if not lvl and player:FindFirstChild("leaderstats") then
        local stats = player.leaderstats
        if stats:FindFirstChild("Level") then lvl = stats.Level.Value
        elseif stats:FindFirstChild("Lvl") then lvl = stats.Lvl.Value end
    end
    return lvl or "?"
end

-- Active ESP tracking dictionary
local activeEspLabels = {}

UserInputService.InputBegan:Connect(function(input, gp)
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

    if listeningKey then
        if input.KeyCode ~= Enum.KeyCode.Escape then
            Keybinds[listeningKey] = input.KeyCode
            buttonMap[listeningKey].Text = input.KeyCode.Name
            saveKeybinds()
        else
            buttonMap[listeningKey].Text = Keybinds[listeningKey].Name
        end
        buttonMap[listeningKey].TextColor3 = Color3.fromRGB(0, 255, 128)
        listeningKey = nil
        return
    end

    if input.KeyCode == Keybinds.MenuToggle then 
        toggleMenu() 
        return 
    end

    if gp then return end

    if input.KeyCode == Keybinds.PlayerCam then
        npcCamlockActive = false
        playerCamlockActive = not playerCamlockActive
        currentTarget = playerCamlockActive and getClosestPlayer() or nil
        if playerCamlockActive and not currentTarget then playerCamlockActive = false end
        updateStatusText()
    elseif input.KeyCode == Keybinds.NpcCam then
        playerCamlockActive = false
        npcCamlockActive = not npcCamlockActive
        currentTarget = npcCamlockActive and getClosestNPC() or nil
        if npcCamlockActive and not currentTarget then npcCamlockActive = false end
        updateStatusText()
    elseif input.KeyCode == Keybinds.SuperJump then
        local char = localPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(char.HumanoidRootPart.AssemblyLinearVelocity.X, 180, char.HumanoidRootPart.AssemblyLinearVelocity.Z)
        end
    elseif input.KeyCode == Keybinds.LongDash then
        local char = localPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local look = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z).Unit
            -- Long dash multiplier set to 350
            char.HumanoidRootPart.AssemblyLinearVelocity = look * 350 + Vector3.new(0, 20, 0)
        end
    elseif input.KeyCode == Keybinds.TpEnemy then
        local char = localPlayer.Character
        local tChar = getClosestPlayer()
        if char and char:FindFirstChild("HumanoidRootPart") and tChar and tChar:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = tChar.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
        end
    elseif input.KeyCode == Keybinds.TpAlly then
        local char = localPlayer.Character
        local tChar = getClosestAlly()
        if char and char:FindFirstChild("HumanoidRootPart") and tChar and tChar:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = tChar.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
        end
    end
end)

RunService.RenderStepped:Connect(function(dt)
    -- Camera Lock Loop
    if (playerCamlockActive or npcCamlockActive) and currentTarget and currentTarget:FindFirstChild("HumanoidRootPart") then
        local hum = currentTarget:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then
            playerCamlockActive, npcCamlockActive, currentTarget = false, false, nil
            updateStatusText()
            return
        end
        local targetCenter = currentTarget.HumanoidRootPart.Position + Vector3.new(0, 0.2, 0)
        camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.Position, targetCenter), 1 - math.exp(-35 * dt))
    end

    -- Track current players for ESP
    local currentPlayers = {}

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer then
            currentPlayers[player] = true
            local char = player.Character
            local head = char and char:FindFirstChild("Head")
            local hum = char and char:FindFirstChildOfClass("Humanoid")

            if char and head and hum and hum.Health > 0 then
                -- Create or grab label
                local label = activeEspLabels[player]
                if not label then
                    label = Instance.new("TextLabel")
                    label.BackgroundTransparency = 1
                    label.Font = Enum.Font.SourceSansBold
                    label.TextSize = 13
                    label.TextStrokeTransparency = 0.4
                    label.Size = UDim2.new(0, 200, 0, 45)
                    label.AnchorPoint = Vector2.new(0.5, 1)
                    label.Parent = EspContainer
                    activeEspLabels[player] = label
                end

                -- Team Color Check (Red for Pirates, Blue for Marines)
                local teamColor = Color3.fromRGB(255, 60, 60)
                if player.Team then
                    local teamName = player.Team.Name:lower()
                    if teamName:find("marine") then
                        teamColor = Color3.fromRGB(60, 150, 255)
                    end
                end

                -- Position above the head
                local headPos = head.Position + Vector3.new(0, 3.5, 0)
                local screenPos, onScreen = camera:WorldToViewportPoint(headPos)

                if onScreen and screenPos.Z > 0 then
                    label.Visible = true
                    label.Position = UDim2.new(0, screenPos.X, 0, screenPos.Y)
                    label.TextColor3 = teamColor
                    
                    local lvl = getPlayerLevel(player)
                    local currentHp = math.floor(hum.Health)
                    local maxHp = math.floor(hum.MaxHealth)
                    
                    -- Username & Level on top, Health status below
                    label.Text = string.format("%s [Lvl: %s]\nHP: %d / %d", player.Name, tostring(lvl), currentHp, maxHp)
                else
                    label.Visible = false
                end
            else
                if activeEspLabels[player] then
                    activeEspLabels[player].Visible = false
                end
            end
        end
    end

    -- Cleanup labels for players who left
    for player, label in pairs(activeEspLabels) do
        if not currentPlayers[player] then
            label:Destroy()
            activeEspLabels[player] = nil
        end
    end
end)

print("[Keybind Settings]: Long Dash Set to 350 & Loaded Successfully!")
