-- Safe Module Loader for Fast Attack Remotes
local function SafeWaitForChild(parent, childName)
    local success, result = pcall(function()
        return parent:WaitForChild(childName, 5)
    end)
    return result
end

local Modules = SafeWaitForChild(ReplicatedStorage, "Modules")
local Net = Modules and SafeWaitForChild(Modules, "Net")
local RegisterAttack = Net and SafeWaitForChild(Net, "RE/RegisterAttack")
local RegisterHit = Net and SafeWaitForChild(Net, "RE/RegisterHit")

local function IsAlive(character)
    return character and character:FindFirstChild("Humanoid") and character.Humanoid.Health > 0
end

local function ProcessEnemies(OthersEnemies, Folder)
    local BasePart = nil
    for _, Enemy in ipairs(Folder:GetChildren()) do
        local Head = Enemy:FindFirstChild("Head")
        if Head and IsAlive(Enemy) and localPlayer:DistanceFromCharacter(Head.Position) < FastAttack.Distance then
            if Enemy ~= localPlayer.Character then
                table.insert(OthersEnemies, { Enemy, Head })
                BasePart = Head
            end
        end
    end
    return BasePart
end

function FastAttack:Attack(BasePart, OthersEnemies)
    if not BasePart or #OthersEnemies == 0 or not RegisterAttack or not RegisterHit then return end
    pcall(function()
        RegisterAttack:FireServer(FastAttackSettings.ClickDelay or 0)
        RegisterHit:FireServer(BasePart, OthersEnemies)
    end)
end

function FastAttack:AttackNearest()
    local OthersEnemies = {}
    local EnemiesFolder = Workspace:FindFirstChild("Enemies")
    local CharactersFolder = Workspace:FindFirstChild("Characters")
    local Part1 = nil
    local Part2 = nil
    if EnemiesFolder then Part1 = ProcessEnemies(OthersEnemies, EnemiesFolder) end
    if CharactersFolder then Part2 = ProcessEnemies(OthersEnemies, CharactersFolder) end
    
    local character = localPlayer.Character
    if not character then return end
    
    local equippedWeapon = character:FindFirstChildOfClass("Tool")
    if equippedWeapon then
        for _, enemyData in ipairs(OthersEnemies) do
            local enemy = enemyData[1]
            if enemy and enemy:FindFirstChild("HumanoidRootPart") and character:FindFirstChild("HumanoidRootPart") then
                local direction = (enemy.HumanoidRootPart.Position - character:GetPivot().Position).Unit
                pcall(function()
                    local leftClick = equippedWeapon:FindFirstChild("LeftClickRemote")
                    if leftClick then
                        leftClick:FireServer(direction, 1)
                    end
                end)
            end
        end
    end
    
    if #OthersEnemies > 0 then
        self:Attack(Part1 or Part2, OthersEnemies)
    end
end

function FastAttack:BladeHits()
    local Equipped = IsAlive(localPlayer.Character) and localPlayer.Character:FindFirstChildOfClass("Tool")
    if Equipped and Equipped.ToolTip ~= "Gun" then
        self:AttackNearest()
    end
end

-- Hardcore Anti-Stun & Air-Float Loop
RunService.Stepped:Connect(function()
    if not NoStunSettings.Enabled then return end
    local char = localPlayer.Character
    if not char then return end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    local rootPart = char:FindFirstChild("HumanoidRootPart")
    
    if humanoid then
        pcall(function()
            humanoid.Sit = false
            humanoid.PlatformStand = false
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
            
            local state = humanoid:GetState()
            if state == Enum.HumanoidStateType.Ragdoll or 
               state == Enum.HumanoidStateType.FallingDown or 
               state == Enum.HumanoidStateType.PlatformStanding or 
               state == Enum.HumanoidStateType.Seated then
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end
        end)
    end
    
    if rootPart then
        if rootPart.Anchored then
            rootPart.Anchored = false
        end
        
        -- If No Stun is enabled, player is holding a move, and they are in the air (FreeFall or Jumping state)
        if isHoldingMove and humanoid and (humanoid:GetState() == Enum.HumanoidStateType.Freefall or humanoid:GetState() == Enum.HumanoidStateType.Jumping) then
            rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end
    end
    
    for _, child in ipairs(char:GetDescendants()) do
        if (child:IsA("BodyVelocity") or child:IsA("BodyPosition") or child:IsA("LinearVelocity")) and child.Name ~= "LongDashVelocity" then
            pcall(function() child:Destroy() end)
        end
    end
end)

-- Save keybinds to file function
local function saveKeybinds()
    if writefile and HttpService then
        local encodedTable = {}
        for keyName, keyCode in pairs(Keybinds) do
            encodedTable[keyName] = keyCode.Name
        end
        pcall(function()
            writefile(saveFileName, HttpService:JSONEncode(encodedTable))
        end)
    end
end

-- Load saved keybinds if file exists
local function loadKeybinds()
    if writefile and readfile and isfile and isfile(saveFileName) then
        local success, decoded = pcall(function()
            return HttpService:JSONDecode(readfile(saveFileName))
        end)
        if success and type(decoded) == "table" then
            for keyName, enumName in pairs(decoded) do
                if Keybinds[keyName] ~= nil and Enum.KeyCode[enumName] then
                    Keybinds[keyName] = Enum.KeyCode[enumName]
                end
            end
        end
    end
end

local playerCamlockActive, npcCamlockActive, currentTarget = false, false, nil

local function isPlayerAlly(player)
    if player == localPlayer then return true end
    local success, isAlly = pcall(function()
        if player:GetAttribute("CrewMember") == true then return true end
        if player:GetAttribute("Ally") == true then return true end
        return false
    end)
    if success and isAlly then return true end
    if localPlayer.Team and player.Team and localPlayer.Team == player.Team then
        if localPlayer.Team.Name:lower() ~= "pirates" then return true end
    end
    return false
end

local function getClosestPlayer()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local myRootPos = char.HumanoidRootPart.Position

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and not isPlayerAlly(p) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (myRootPos - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortest then shortest, closest = dist, p.Character end
            end
        end
    end
    return closest
end

local function getClosestAlly()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local myRootPos = char.HumanoidRootPart.Position

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and isPlayerAlly(p) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (myRootPos - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortest then shortest, closest = dist, p.Character end
            end
        end
    end
    return closest
end

local function getClosestNPC()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local myRootPos = char.HumanoidRootPart.Position

    for _, obj in ipairs(enemiesFolder:GetChildren()) do
        if obj:IsA("Model") and obj ~= char then
            local hum, root = obj:FindFirstChildOfClass("Humanoid"), obj:FindFirstChild("HumanoidRootPart")
            if hum and root and not Players:GetPlayerFromCharacter(obj) and hum.Health > 0 then
                local dist = (myRootPos - root.Position).Magnitude
                if dist < shortest then shortest, closest = dist, obj end
            end
        end
    end
    return closest
end

local function updateStatusText()
    if playerCamlockActive then StatusLabel.Text, StatusLabel.TextColor3 = "Cam: Player ON", Color3.fromRGB(0, 255, 128)
    elseif npcCamlockActive then StatusLabel.Text, StatusLabel.TextColor3 = "Cam: NPC ON", Color3.fromRGB(0, 255, 128)
    else StatusLabel.Text, StatusLabel.TextColor3 = "Cam: OFF", Color3.fromRGB(255, 50, 50) end
end

-- Team Color Resolver for ESP
local function getPlayerColor(player)
    local team = player.Team
    if team and team.Name:lower():find("marine") then
        return Color3.fromRGB(0, 150, 255)
    end
    return Color3.fromRGB(255, 50, 50)
end

-- Clean Billboard ESP Setup
local function createESP(player)
    if player == localPlayer then return end

    local function setupCharacter(char)
        if activeESPs[player] then
            activeESPs[player].billboard:Destroy()
            activeESPs[player].conn:Disconnect()
            activeESPs[player] = nil
        end

        local head = char:WaitForChild("Head", 5)
        local humanoid = char:WaitForChild("Humanoid", 5)
        if not head or not humanoid then return end

        local billboard = Instance.new("BillboardGui")
        billboard.Name = "PlayerESP"
        billboard.Size = UDim2.new(0, 180, 0, 45)
        billboard.StudsOffset = Vector3.new(0, 2.5, 0)
        billboard.AlwaysOnTop = true
        billboard.Adornee = head
        billboard.Parent = CoreGui

        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextSize = 13
        textLabel.TextColor3 = getPlayerColor(player)
        textLabel.TextStrokeTransparency = 0
        textLabel.TextXAlignment = Enum.TextXAlignment.Center
        textLabel.TextYAlignment = Enum.TextYAlignment.Center
        textLabel.Parent = billboard

        local elapsed = 0
        local connection
        connection = RunService.Heartbeat:Connect(function(dt)
            elapsed = elapsed + dt
            if elapsed < 0.2 then return end
            elapsed = 0

            if not char or not char.Parent or humanoid.Health <= 0 then
                billboard:Destroy()
                connection:Disconnect()
                activeESPs[player] = nil
                return
            end

            local lvl = "N/A"
            if player:FindFirstChild("leaderstats") and player.leaderstats:FindFirstChild("Level") then
                lvl = player.leaderstats.Level.Value
            elseif player:FindFirstChild("Data") and player.Data:FindFirstChild("Level") then
                lvl = player.Data.Level.Value
            end

            textLabel.TextColor3 = getPlayerColor(player)
            textLabel.Text = string.format("%s [Lvl. %s]\nHP: %d/%d", player.Name, tostring(lvl), math.floor(humanoid.Health), math.floor(humanoid.MaxHealth))
        end)

        activeESPs[player] = {billboard = billboard, conn = connection}
    end

    if player.Character then
        setupCharacter(player.Character)
    end
    player.CharacterAdded:Connect(setupCharacter)
end

for _, p in ipairs(Players:GetPlayers()) do
    createESP(p)
end
Players.PlayerAdded:Connect(createESP)

    end
end)

-- Inputs
UserInputService.InputBegan:Connect(function(input, gp)
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

    -- Ignore key inputs if the user is currently typing in chat or another textbox
    if UserInputService:GetFocusedTextBox() then return end

    if listeningKey then
        if input.KeyCode ~= Enum.KeyCode.Escape then
            Keybinds[listeningKey] = input.KeyCode
            buttonMap[listeningKey].Text = input.KeyCode.Name
            saveKeybinds()
        else
            buttonMap[listeningKey].Text = Keybinds[listeningKey].Name
        end
        buttonMap[listeningKey].TextColor3 = Color3.fromRGB(0, 255, 128)
        listeningKey = nil
        return
    end

    if input.KeyCode == Keybinds.MenuToggle then 
        toggleMenu() 
        return 
    end

    if gp then return end

    if input.KeyCode == Keybinds.PlayerCam then
        npcCamlockActive = false
        playerCamlockActive = not playerCamlockActive
        currentTarget = playerCamlockActive and getClosestPlayer() or nil
        if playerCamlockActive and not currentTarget then playerCamlockActive = false end
        updateStatusText()
    elseif input.KeyCode == Keybinds.NpcCam then
        playerCamlockActive = false
        npcCamlockActive = not npcCamlockActive
        currentTarget = npcCamlockActive and getClosestNPC() or nil
        if npcCamlockActive and not currentTarget then npcCamlockActive = false end
        updateStatusText()
    elseif input.KeyCode == Keybinds.SuperJump then
        local char = localPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local root = char.HumanoidRootPart
            root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, 180, root.AssemblyLinearVelocity.Z)
        end
    elseif input.KeyCode == Keybinds.LongDash then
        local char = localPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local root = char.HumanoidRootPart
            local look = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z).Unit
            root.AssemblyLinearVelocity = look * 350 + Vector3.new(0, 20, 0)
        end
    elseif input.KeyCode == Keybinds.TpEnemy then
        local char = localPlayer.Character
        local tChar = getClosestPlayer()
        if char and char:FindFirstChild("HumanoidRootPart") and tChar and tChar:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = tChar.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
        end
    elseif input.KeyCode == Keybinds.TpAlly then
        local char = localPlayer.Character
        local tChar = getClosestAlly()
        if char and char:FindFirstChild("HumanoidRootPart") and tChar and tChar:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = tChar.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
        end
    end
end)

-- Main Camera Lock Loop

    if (playerCamlockActive or npcCamlockActive) and currentTarget and currentTarget:FindFirstChild("HumanoidRootPart") then
        local humanoid = currentTarget:FindFirstChildOfClass("Humanoid")
        local targetPlayer = Players:GetPlayerFromCharacter(currentTarget)
        
        if not humanoid or humanoid.Health <= 0 or (playerCamlockActive and targetPlayer and isPlayerAlly(targetPlayer)) then
            playerCamlockActive = false
            npcCamlockActive = false
            currentTarget = nil
            updateStatusText()
            return
        end

        local targetRoot = currentTarget.HumanoidRootPart
        local currentCamCFrame = camera.CFrame
        local targetCenter = targetRoot.Position + Vector3.new(0, 0.2, 0)
        local idealCFrame = CFrame.new(currentCamCFrame.Position, targetCenter)
        
        camera.CFrame = currentCamCFrame:Lerp(idealCFrame, 1 - math.exp(-35 * dt))
    end
end)
