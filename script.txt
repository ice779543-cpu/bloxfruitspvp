-- Safe Module Loader
local function SafeWaitForChild(parent, childName)
    local success, result = pcall(function()
        return parent:WaitForChild(childName, 5)
    end)
    return result
end

local Modules = SafeWaitForChild(ReplicatedStorage, "Modules")
local Net = Modules and SafeWaitForChild(Modules, "Net")
local RegisterAttack = Net and SafeWaitForChild(Net, "RE/RegisterAttack")
local RegisterHit = Net and SafeWaitForChild(Net, "RE/RegisterHit")

-- Dedicated Attack Function for Mobs
local function executeAttack(mobRoot, mobHead)
    local char = localPlayer.Character
    if not char then return end
    
    local equippedWeapon = char:FindFirstChildOfClass("Tool")
    if equippedWeapon then
        pcall(function()
            local leftClick = equippedWeapon:FindFirstChild("LeftClickRemote")
            if leftClick and mobRoot then
                local direction = (mobRoot.Position - char:GetPivot().Position).Unit
                leftClick:FireServer(direction, 1)
            end
        end)
    end
    
    if RegisterAttack and RegisterHit and mobHead then
        pcall(function()
            RegisterAttack:FireServer(0)
            RegisterHit:FireServer(mobHead, { {mobHead.Parent, mobHead} })
        end)
    end
end

-- Save / Load Functions
local function saveKeybinds()
    if writefile and HttpService then
        local encodedTable = {}
        for keyName, keyCode in pairs(Keybinds) do encodedTable[keyName] = keyCode.Name end
        pcall(function() writefile(saveFileName, HttpService:JSONEncode(encodedTable)) end)
    end
end

local function loadKeybinds()
    if writefile and readfile and isfile and isfile(saveFileName) then
        local success, decoded = pcall(function() return HttpService:JSONDecode(readfile(saveFileName)) end)
        if success and type(decoded) == "table" then
            for keyName, enumName in pairs(decoded) do
                if Keybinds[keyName] ~= nil and Enum.KeyCode[enumName] then Keybinds[keyName] = Enum.KeyCode[enumName] end
            end
        end
    end
end
loadKeybinds()

-- Fast Attack Loop Thread
task.spawn(function()
    while true do
        if FastAttackSettings.AutoClick then
            local char = localPlayer.Character
            if char and char:FindFirstChildOfClass("Tool") then
                local enemiesFolder = Workspace:FindFirstChild("Enemies")
                if enemiesFolder then
                    for _, mob in pairs(enemiesFolder:GetChildren()) do
                        local head = mob:FindFirstChild("Head")
                        local root = mob:FindFirstChild("HumanoidRootPart")
                        local hum = mob:FindFirstChild("Humanoid")
                        if head and root and hum and hum.Health > 0 and localPlayer:DistanceFromCharacter(root.Position) <= FastAttack.Distance then
                            executeAttack(root, head)
                        end
                    end
                end
            end
        end
        task.wait(FastAttackSettings.ClickDelay)
    end
end)

-- Auto V3 / V4 background loops
task.spawn(function()

        task.wait(1)
        pcall(function()
            if PvPStates.AutoV3 then
                ReplicatedStorage.Remotes.CommF_:InvokeServer("ActivateAbility")
            end
            if RaidStates.AutoV4 then
                ReplicatedStorage.Remotes.CommF_:InvokeServer("Awakening")
            end
        end)
    end
end)

-- Raid Automation Loop (Smooth Flying, Hovering, & Fast Attack Integration)
local currentTween = nil
local function stopTween()
    if currentTween then
        currentTween:Cancel()
        currentTween = nil
    end
end

local function smoothFlyTo(targetCFrame, speed)
    local char = localPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    stopTween()
    local distance = (root.Position - targetCFrame.Position).Magnitude
    local timeTaken = distance / (speed or 250)
    
    local tweenInfo = TweenInfo.new(timeTaken, Enum.EasingStyle.Linear)
    currentTween = TweenService:Create(root, tweenInfo, {CFrame = targetCFrame})
    currentTween:Play()
end

task.spawn(function()
    while true do
        task.wait(0.2)
        pcall(function()
            if RaidStates.AutoFarmRaid then
                local playerGui = localPlayer:FindFirstChild("PlayerGui")
                local raidTimer = playerGui and playerGui:FindFirstChild("Main") and playerGui.Main:FindFirstChild("TopHUDList") and playerGui.Main.TopHUDList:FindFirstChild("RaidTimer")
                
                -- 1. Auto Start Raid if timer is hidden/inactive
                if raidTimer and raidTimer.Visible == false then
                    local function GetBP(itemName)
                        for _, v in ipairs(localPlayer.Backpack:GetChildren()) do
                            if v.Name == itemName then return v end
                        end
                        if localPlayer.Character and localPlayer.Character:FindFirstChild(itemName) then
                            return localPlayer.Character[itemName]
                        end
                        return nil
                    end

                    if GetBP("Special Microchip") then
                        local World2 = Workspace:FindFirstChild("Map") and Workspace.Map:FindFirstChild("CircleIsland")
                        if World2 then
                            smoothFlyTo(CFrame.new(-6438.73535, 250.645355, -4501.50684), 300)
                            task.wait(1)
                            local btn = Workspace.Map.CircleIsland:FindFirstChild("RaidSummon2")
                            if btn and btn:FindFirstChild("Button") and btn.Button:FindFirstChild("Main") and btn.Button.Main:FindFirstChild("ClickDetector") then
                                fireclickdetector(btn.Button.Main.ClickDetector)
                            end
                        else
                            pcall(function()
                                ReplicatedStorage.Remotes.CommF_:InvokeServer("requestEntrance", Vector3.new(-5097.93164, 316.447021, -3142.66602))
                            end)
                            task.wait(0.5)
                            smoothFlyTo(CFrame.new(-5033.50879, 315.014252, -2947.77539), 300)
                            local boatCastle = Workspace:FindFirstChild("Map") and Workspace.Map:FindFirstChild("Boat Castle")
                            if boatCastle and boatCastle:FindFirstChild("RaidSummon2") then
                                local clickDet = boatCastle.RaidSummon2:FindFirstChild("Button") and boatCastle.RaidSummon2.Button:FindFirstChild("Main") and boatCastle.RaidSummon2.Button.Main:FindFirstChild("ClickDetector")
                                if clickDet then fireclickdetector(clickDet) end
                            end
                        end
                    end
                end

                -- 2. In-Raid Behavior: Hover above mobs and rapidly execute attacks
                if raidTimer and raidTimer.Visible == true then
                    local enemiesFolder = Workspace:FindFirstChild("Enemies")
                    local targetMobRoot, targetMobHead = nil, nil
                    
                    if enemiesFolder then
                        for _, mob in pairs(enemiesFolder:GetChildren()) do
                            local hum = mob:FindFirstChild("Humanoid")
                            local root = mob:FindFirstChild("HumanoidRootPart")
                            local head = mob:FindFirstChild("Head")
                            if hum and root and head and hum.Health > 0 then
                                targetMobRoot = root
                                targetMobHead = head
                                break
                            end
                        end
                    end

                    local character = localPlayer.Character
                    local myRoot = character and character:FindFirstChild("HumanoidRootPart")

                    if targetMobRoot and targetMobHead and myRoot then
                        stopTween()
                        myRoot.Anchored = false
                        myRoot.CFrame = targetMobRoot.CFrame * CFrame.new(0, 15, 0)
                        myRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                        myRoot.Anchored = true

                        -- Trigger fast attack continuously while hovering above the mob
                        executeAttack(targetMobRoot, targetMobHead)
                    else
                        local character = localPlayer.Character
                        local myRoot = character and character:FindFirstChild("HumanoidRootPart")
                        if myRoot then myRoot.Anchored = false end

                        -- If no enemies are left on this island, move to the next island
                        local locs = Workspace:FindFirstChild("_WorldOrigin") and Workspace._WorldOrigin:FindFirstChild("Locations")
                        if locs then
                            local order = {5, 4, 3, 2, 1}
                            for _, id in ipairs(order) do
                                local islandName = "Island " .. id
                                local island = locs:FindFirstChild(islandName)
                                if island and myRoot then
                                    if (island.Position - myRoot.Position).Magnitude <= 4500 then
                                        smoothFlyTo(island.CFrame * CFrame.new(0, 40, 0), 300)
                                        break
                                    end
                                end
                            end
                        end
                    end
                else
                    local character = localPlayer.Character
                    local myRoot = character and character:FindFirstChild("HumanoidRootPart")
                    if myRoot then myRoot.Anchored = false end
                end
            end
        end)
    end
end)
