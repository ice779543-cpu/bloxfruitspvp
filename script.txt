local function IsBlacklisted(name)

local function isAlly(plr)
    if not MiscSettings.AntiAlly then return false end
    if plr == localPlayer then return true end
    
    local success, result = pcall(function()
        local myData = localPlayer:FindFirstChild("Data")
        local targetData = plr:FindFirstChild("Data")
        
        if myData and targetData then
            local myCrew = myData:FindFirstChild("Crew") or myData:FindFirstChild("Alliance")
            local targetCrew = targetData:FindFirstChild("Crew") or targetData:FindFirstChild("Alliance")
            if myCrew and targetCrew and myCrew.Value ~= "" and myCrew.Value == targetCrew.Value then
                return true
            end
        end

        if localPlayer.Team and plr.Team and localPlayer.Team == plr.Team then
            if localPlayer.Team.Name ~= "Pirates" then
                return true
            end
        end

        return false
    end)

    if success then return result end
    return false
end

local function canGetMyBounty(plr)
    if plr == localPlayer then return false end
    if isAlly(plr) then return false end

    local success, result = pcall(function()
        local myTeam = localPlayer.Team and localPlayer.Team.Name or ""
        local targetTeam = plr.Team and plr.Team.Name or ""

        if myTeam == "Marines" and targetTeam == "Marines" then
            return false
        end

        local myData = localPlayer:FindFirstChild("Data")
        local targetData = plr:FindFirstChild("Data")

        if myData and targetData then
            local myLevel = myData:FindFirstChild("Level") and myData.Level.Value or 0
            local targetLevel = targetData:FindFirstChild("Level") and targetData.Level.Value or 0
            
            if math.abs(myLevel - targetLevel) > 350 and myLevel > targetLevel then
                return false
            end
        end

        local char = plr.Character
        local hum = char and char:FindFirstChild("Humanoid")
        if not hum or hum.Health <= 0 then return false end

        return true
    end)

    if success then return result end
    return false
end

local function findAimedNpcTarget()

    local shortestDist = math.huge
    local screenSize = camera.ViewportSize
    local screenCenter = Vector2.new(screenSize.X / 2, screenSize.Y / 2)

    local enemiesFolder = Workspace:FindFirstChild("Enemies")
    if enemiesFolder then
        for _, mob in ipairs(enemiesFolder:GetChildren()) do
            if isValidMob(mob) then
                local root = mob:FindFirstChild("HumanoidRootPart") or mob:FindFirstChild("Head")
                if root then
                    local screenPos, onScreen = camera:WorldToViewportPoint(root.Position)
                    if onScreen then
                        local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                        if screenDist < shortestDist then
                            shortestDist = screenDist
                            bestTarget = root
                        end
                    end
                end
            end
        end
    end
    return bestTarget
end

local function executeMultiAttack(targetsList)
    pcall(function()
        local char = localPlayer.Character
        if not char then return end
        local equippedWeapon = char:FindFirstChildOfClass("Tool")
        if equippedWeapon then
            local attackRemote = equippedWeapon:FindFirstChild("LeftClickRemote") or equippedWeapon:FindFirstChild("ToolAttack") or equippedWeapon:FindFirstChild("AttackRemote")
            if attackRemote and typeof(attackRemote.FireServer) == "function" and #targetsList > 0 then
                local firstRoot = targetsList[1].Root
                local direction = (firstRoot.Position - char:GetPivot().Position).Unit
                attackRemote:FireServer(direction, 1)
            end
        end
        
        local regAttack = getRegisterAttack()
        if regAttack and typeof(regAttack.FireServer) == "function" then regAttack:FireServer(0) end
        
        local regHit = getRegisterHit()
        if regHit and typeof(regHit.FireServer) == "function" and #targetsList > 0 then
            local hitParams = {}
            for _, data in ipairs(targetsList) do
                if data.Head and data.Head.Parent then
                    table.insert(hitParams, {data.Head.Parent, data.Head})
                end
            end
            if #hitParams > 0 then
                regHit:FireServer(targetsList[1].Head, hitParams)
            end
        end
    end)
end

local function equipToolType(toolType)
    local char = localPlayer.Character
    if not char then return false end

    local currentTool = char:FindFirstChildOfClass("Tool")
    if currentTool then
        local tip = currentTool.ToolTip:lower()
        if tip == toolType:lower() or (toolType == "Fruit" and tip:find("blox fruit")) then
            return true
        end
    end

    local foundTool = nil
    local containerList = {localPlayer.Backpack, char}

    for _, container in ipairs(containerList) do
        for _, t in ipairs(container:GetChildren()) do
            if t:IsA("Tool") then
                local tip = t.ToolTip:lower()
                local name = t.Name:lower()
                
                if toolType == "Sword" and (tip == "sword" or name:find("sword")) then
                    foundTool = t; break
                elseif toolType == "Fruit" and (tip:find("blox fruit") or tip == "fruit" or name:find("fruit")) then
                    foundTool = t; break
                elseif toolType == "Gun" and (tip == "gun" or name:find("gun") or name:find("rifle")) then
                    foundTool = t; break
                elseif toolType == "Melee" and (tip == "blox fighting style" or tip == "melee" or name:find("fighting") or name:find("step")) then
                    foundTool = t; break
                end
            end
        end
        if foundTool then break end
    end

    if foundTool then
        if foundTool.Parent == localPlayer.Backpack then
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid then
                humanoid:EquipTool(foundTool)
                task.wait(0.3)
                return true
            end
        else
            return true
        end
    end
    return false
end

-- Enhanced target finder with hardcoded fallback position for unloaded islands
local function findNearestMaterialNPC()
    local bestTargetPos = nil
    local shortestDist = math.huge
    local char = localPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end

    local matInfo = availableMaterials[AutoFarmMaterialSettings.SelectedMaterialIndex]
    if not matInfo or not matInfo.Mobs then return nil end

    -- 1. Check loaded mobs in workspace
    local enemiesFolder = Workspace:FindFirstChild("Enemies")
    if enemiesFolder then
        for _, mob in ipairs(enemiesFolder:GetChildren()) do
            if isValidMob(mob) then
                local match = false
                for _, mobName in ipairs(matInfo.Mobs) do
                    if mob.Name:lower():find(mobName:lower()) then
                        match = true
                        break
                    end
                end

                if match then
                    local mobRoot = mob:FindFirstChild("HumanoidRootPart")
                    if mobRoot then
                        local dist = (mobRoot.Position - root.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            bestTargetPos = mobRoot.Position
                        end
                    end
                end
            end
        end
    end

    -- 2. Fallback to hardcoded coordinates if mob is on another island and not spawned yet
    if not bestTargetPos then
        for _, mobName in ipairs(matInfo.Mobs) do
            for key, spawnPos in pairs(MobSpawnLocations) do
                if key:lower() == mobName:lower() then
                    bestTargetPos = spawnPos
                    break
                end
            end
            if bestTargetPos then break end
        end
    end

    return bestTargetPos
end

local function manageAutoFarmFlight()
    pcall(function()
        local char = localPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChild("Humanoid")
        if not root or not hum or hum.Health <= 0 then return end

        local bv = root:FindFirstChild("AutoFarmFlyBV")
        local bg = root:FindFirstChild("AutoFarmFlyBG")

        if AutoFarmMasterySettings.Enabled then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end

            if not bv then
                bv = Instance.new("BodyVelocity")
                bv.Name = "AutoFarmFlyBV"
                bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                bv.Velocity = Vector3.new(0, 0, 0)
                bv.Parent = root
            end
            if not bg then
                bg = Instance.new("BodyGyro")
                bg.Name = "AutoFarmFlyBG"
                bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                bg.CFrame = root.CFrame
                bg.Parent = root
            end

            local targetRoot = findClosestTargetMob()
            if targetRoot and targetRoot.Parent then

local function findCurrentRaidIsland(rootPos)
    local bestTarget = nil
    local shortestDist = math.huge

    local enemiesFolder = Workspace:FindFirstChild("Enemies")
    if enemiesFolder then
        for _, mob in ipairs(enemiesFolder:GetChildren()) do
            local hum = mob:FindFirstChild("Humanoid")
            local root = mob:FindFirstChild("HumanoidRootPart")
            if hum and root and hum.Health > 0 then
                local dist = (root.Position - rootPos).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    bestTarget = root.Position + Vector3.new(0, 20, 0)
                end
            end
        end
    end

    if bestTarget then return bestTarget end
    return nil
end

local function saveSettings()
    pcall(function()
        if writefile and HttpService then
            local encodedTable = {
                Keybinds = {},
                MacroSelections = MacroSelections,
                MacroDelays = MacroDelays,
                MiscSettings = MiscSettings,
                SavedMusicList = SavedMusicList
            }
            for keyName, keyCode in pairs(Keybinds) do 
                encodedTable.Keybinds[keyName] = keyCode.Name 
            end
            writefile(saveFileName, HttpService:JSONEncode(encodedTable))
        end
    end)
end

local function loadSettings()
    pcall(function()
        if readfile and isfile and isfile(saveFileName) then
            local success, decoded = pcall(function() return HttpService:JSONDecode(readfile(saveFileName)) end)
            if success and type(decoded) == "table" then
                if decoded.Keybinds then
                    for keyName, enumName in pairs(decoded.Keybinds) do
                        if Keybinds[keyName] ~= nil and Enum.KeyCode[enumName] then 
                            Keybinds[keyName] = Enum.KeyCode[enumName] 
                        end
                    end
                end
                if decoded.MacroSelections and type(decoded.MacroSelections) == "table" then
                    for i = 1, #MacroSelections do
                        if decoded.MacroSelections[i] then MacroSelections[i] = decoded.MacroSelections[i] end
                    end
                end
                if decoded.MacroDelays and type(decoded.MacroDelays) == "table" then
                    for i = 1, #MacroDelays do
                        if decoded.MacroDelays[i] then MacroDelays[i] = decoded.MacroDelays[i] end
                    end
                end
                if decoded.MiscSettings and type(decoded.MiscSettings) == "table" then
                    for k, v in pairs(decoded.MiscSettings) do
                        if MiscSettings[k] ~= nil then MiscSettings[k] = v end
                    end
                end
                if decoded.SavedMusicList and type(decoded.SavedMusicList) == "table" then
                    SavedMusicList = decoded.SavedMusicList
                end
            end
        end
    end)
end
loadSettings()

    local Row = Instance.new("Frame")

    Label.Parent = Row

    local SliderBar = Instance.new("TextButton")
    SliderBar.Size = UDim2.new(1, -20, 0, 14)
    SliderBar.Position = UDim2.new(0, 10, 0, 30)
    SliderBar.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
    SliderBar.Text = ""
    SliderBar.Parent = Row
    Instance.new("UICorner", SliderBar).CornerRadius = UDim.new(0, 4)

    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new((currentVal - minVal) / (maxVal - minVal), 0, 1, 0)
    Fill.BackgroundColor3 = Color3.fromRGB(0, 255, 128)
    Fill.BorderSizePixel = 0
    Fill.Parent = SliderBar
    Instance.new("UICorner", Fill).CornerRadius = UDim.new(0, 4)

    local dragging = false
    SliderBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = true end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local mousePos = UserInputService:GetMouseLocation()
            local relX = math.clamp((mousePos.X - SliderBar.AbsolutePosition.X) / SliderBar.AbsoluteSize.X, 0, 1)
            local val = math.floor(minVal + ((maxVal - minVal) * relX))
            Fill.Size = UDim2.new(relX, 0, 1, 0)
            Label.Text = name .. ": " .. val
            callback(val)
        end
    end)
end

        MacroSelections[index] = MacroActions[currentIndex]
        SelectButton.Text = MacroSelections[index]
        saveSettings()
    end)

    local DelayButton = Instance.new("TextButton")
    DelayButton.Size = UDim2.new(0, 95, 0, 26)

    DelayButton.Text = "Delay: " .. string.format("%.1fs", MacroDelays[index])
    DelayButton.TextColor3 = Color3.fromRGB(255, 200, 0)
    DelayButton.Parent = Row
    Instance.new("UICorner", DelayButton).CornerRadius = UDim.new(0, 4)

                currentIdx = idx
                break
            end
        end
        currentIdx = currentIdx + 1
        if currentIdx > #delaySteps then currentIdx = 1 end
        MacroDelays[index] = delaySteps[currentIdx]
        DelayButton.Text = "Delay: " .. string.format("%.1fs", MacroDelays[index])
        saveSettings()
    end)
end

task.spawn(function()
    while true do
        pcall(function()
            if FastAttackSettings.AutoClick or AutoFarmMasterySettings.Enabled or AutoFarmHauntedSettings.Enabled or AutoFarmMaterialSettings.Enabled then
                local char = localPlayer.Character
                if char and char:FindFirstChildOfClass("Tool") then
                    local myRoot = char:FindFirstChild("HumanoidRootPart")
                    if myRoot then
                        local targetsList = {}
                        
                        local enemiesFolder = Workspace:FindFirstChild("Enemies")
                        if enemiesFolder then
                            for _, mob in pairs(enemiesFolder:GetChildren()) do
                                if isValidMob(mob) then
                                    local head = mob:FindFirstChild("Head")
                                    local root = mob:FindFirstChild("HumanoidRootPart")
                                    local hum = mob:FindFirstChild("Humanoid")
                                    if head and root and hum and hum.Health > 0 and (root.Position - myRoot.Position).Magnitude <= 120 then
                                        table.insert(targetsList, {Head = head, Root = root})
                                    end
                                end
                            end
                        end

                        for _, plr in ipairs(Players:GetPlayers()) do
                            if plr ~= localPlayer and not isAlly(plr) and plr.Character then
                                local pChar = plr.Character
                                local pHum = pChar:FindFirstChild("Humanoid")
                                local pRoot = pChar:FindFirstChild("HumanoidRootPart")
                                local pHead = pChar:FindFirstChild("Head")
                                
                                if pHum and pHum.Health > 0 and pRoot and pHead then
                                    if (pRoot.Position - myRoot.Position).Magnitude <= 120 then
                                        table.insert(targetsList, {Head = pHead, Root = pRoot})
                                    end
                                end
                            end
                        end

                        if #targetsList > 0 then
                            executeMultiAttack(targetsList)
                        end
                    end
                end
            end
        end)
        task.wait(FastAttackSettings.ClickDelay)
    end
end)

RunService.RenderStepped:Connect(function()
    pcall(function()
        manageWaterPlatform()

        if NpcHitboxSettings.Enabled then
            local enemiesFolder = Workspace:FindFirstChild("Enemies")
            if enemiesFolder then
                for _, mob in ipairs(enemiesFolder:GetChildren()) do
                    if isValidMob(mob) then
                        for _, part in ipairs(mob:GetDescendants()) do
                            if part:IsA("BasePart") and not part:IsA("Accessory") and not part.Parent:IsA("Accessory") then
                                if not originalHitboxSizes[part] then
                                    originalHitboxSizes[part] = {
                                        size = part.Size,
                                        transparency = part.Transparency,
                                        material = part.Material,
                                        canCollide = part.CanCollide
                                    }
                                end
                                local safeSize = math.clamp(NpcHitboxSettings.Size, 1, 100)
                                part.Size = Vector3.new(safeSize, safeSize, safeSize)
                                part.Transparency = NpcHitboxSettings.Transparency / 100
                                part.Material = NpcHitboxSettings.Transparency >= 100 and Enum.Material.SmoothPlastic or Enum.Material.Neon
                                part.Color = Color3.fromRGB(255, 100, 100)
                                part.CanCollide = false
                            end
                        end
                    end
                end
            end
        else
            restoreHitboxes()
        end

        if PlayerHitboxSettings.Enabled then
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= localPlayer and not isAlly(plr) and plr.Character then

                    local rootPart = plr.Character:FindFirstChild("HumanoidRootPart")
                    if hum and hum.Health > 0 and rootPart then
                        if not originalPlayerHitboxSizes[rootPart] then
                            originalPlayerHitboxSizes[rootPart] = {
                                size = rootPart.Size,
                                transparency = rootPart.Transparency,
                                material = rootPart.Material,
                                canCollide = rootPart.CanCollide
                            }
                        end
                        local safeSize = math.clamp(PlayerHitboxSettings.Size, 1, 100)
                        rootPart.Size = Vector3.new(safeSize, safeSize, safeSize)
                        rootPart.Transparency = PlayerHitboxSettings.Transparency / 100
                        rootPart.Material = PlayerHitboxSettings.Transparency >= 100 and Enum.Material.SmoothPlastic or Enum.Material.Neon
                        rootPart.Color = Color3.fromRGB(100, 150, 255)
                        rootPart.CanCollide = false
                    end
                end
            end
        else
            restorePlayerHitboxes()
        end

        if AutoRaidSettings.Enabled then
            manageRaidFlight()
        end

        if AutoFarmMasterySettings.Enabled then
            manageAutoFarmFlight()
        end

        if AutoFarmHauntedSettings.Enabled then
            manageHauntedFarmFlight()
        end

        if AutoFarmMaterialSettings.Enabled then
            manageMaterialFarmFlight()
        end

        if MiscSettings.AntiBountyLose then
            manageAntiBountyFlight()
        end

        if CamStates.PlayerCamActive then
            if CamStates.LockedPlayerTarget and CamStates.LockedPlayerTarget.Parent then
                local character = CamStates.LockedPlayerTarget.Parent
                local plr = Players:GetPlayerFromCharacter(character)
                local hum = character:FindFirstChild("Humanoid")
                if plr and isAlly(plr) then
                    CamStates.PlayerCamActive = false
                    CamStates.LockedPlayerTarget = nil
                elseif hum and hum.Health > 0 then
                    camera.CFrame = CFrame.new(camera.CFrame.Position, CamStates.LockedPlayerTarget.Position)
                else
                    CamStates.PlayerCamActive = false
                    CamStates.LockedPlayerTarget = nil
                end
            else
                CamStates.PlayerCamActive = false
                CamStates.LockedPlayerTarget = nil
            end
        end

        if CamStates.NpcCamActive then
            if CamStates.LockedNpcTarget and CamStates.LockedNpcTarget.Parent then
                local hum = CamStates.LockedNpcTarget.Parent:FindFirstChild("Humanoid")
                if hum and hum.Health > 0 then
                    camera.CFrame = CFrame.new(camera.CFrame.Position, CamStates.LockedNpcTarget.Position)
                else
                    CamStates.NpcCamActive = false
                    CamStates.LockedNpcTarget = nil
                end
            else
                CamStates.NpcCamActive = false
                CamStates.LockedNpcTarget = nil
            end
        end

        if ESPStates.Enabled then
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= localPlayer then
                    local char = plr.Character
                    local root = char and char:FindFirstChild("HumanoidRootPart")
                    local hum = char and char:FindFirstChild("Humanoid")
                    local existingEsp = espFolder:FindFirstChild("ESP_" .. plr.Name)
                    
                    if char and root and hum and hum.Health > 0 then
                        local playerLevel = "1"
                        local stats = plr:FindFirstChild("Data") or plr:FindFirstChild("leaderstats")
                        if stats and stats:FindFirstChild("Level") then playerLevel = tostring(stats.Level.Value) end

                        local espColor = Color3.fromRGB(255, 50, 50)
                        if isAlly(plr) then
                            espColor = Color3.fromRGB(0, 255, 128)
                        elseif plr.Team then
                            local teamName = tostring(plr.Team.Name):lower()
                            if teamName:find("marine") then
                                espColor = Color3.fromRGB(50, 150, 255)
                            end
                        end

                        local billboard = existingEsp
                        if not billboard then
                            billboard = Instance.new("BillboardGui")
                            billboard.Name = "ESP_" .. plr.Name
                            billboard.Size = UDim2.new(0, 150, 0, 40)
                            billboard.StudsOffset = Vector3.new(0, 3, 0)
                            billboard.AlwaysOnTop = true
                            billboard.Parent = espFolder
                            
                            local textLbl = Instance.new("TextLabel")
                            textLbl.Name = "TextInfo"
                            textLbl.Size = UDim2.new(1, 0, 1, 0)
                            textLbl.BackgroundTransparency = 1
                            textLbl.Font = Enum.Font.SourceSansBold
                            textLbl.TextSize = 13
                            textLbl.TextStrokeTransparency = 0.2
                            textLbl.Parent = billboard
                        end

                        billboard.Adornee = root
                        local textLbl = billboard:FindFirstChild("TextInfo")
                        if textLbl then
                            textLbl.TextColor3 = espColor
                            textLbl.Text = string.format("%s [Lvl: %s]\n[%d / %d HP]", plr.Name, playerLevel, math.floor(hum.Health), math.floor(hum.MaxHealth))
                        end
                    else
                        if existingEsp then existingEsp:Destroy() end
                    end
                end
            end
        else
            for _, v in ipairs(espFolder:GetChildren()) do v:Destroy() end
        end
        
        if FOVSettings.Enabled then
            local hue = (tick() * FOVSettings.RainbowSpeed) % 1
            FOVCircle.Color = Color3.fromHSV(hue, 1, 1)

            local screenSize = camera.ViewportSize

            FOVCircle.Position = screenCenter
            FOVCircle.Visible = MainFrame.Visible == false

            local target = nil
            local shortestDist = math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= localPlayer and not isAlly(plr) and plr.Character then
                    local root = plr.Character:FindFirstChild("HumanoidRootPart")
                    local hum = plr.Character:FindFirstChild("Humanoid")
                    if root and hum and hum.Health > 0 then
                        local screenPos, onScreen = camera:WorldToViewportPoint(root.Position)
                        if onScreen then
                            local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                            if screenDist <= FOVSettings.Radius and screenDist < shortestDist then
                                shortestDist = screenDist
                                target = root
                            end
                        end
                    end
                end
            end

            if target then
                camera.CFrame = CFrame.new(camera.CFrame.Position, target.Position)
            end
        else
            FOVCircle.Visible = false
        end
    end)
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    pcall(function()
        local char = localPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")

        if input.KeyCode == Keybinds.MenuToggle then
            MainFrame.Visible = not MainFrame.Visible
        elseif input.KeyCode == Keybinds.MenuHide then
            ScreenGui.Enabled = not ScreenGui.Enabled
        elseif input.KeyCode == Keybinds.MacroKey then
            runMacroSequence()
        elseif input.KeyCode == Keybinds.AimbotKey then
            FOVSettings.Enabled = not FOVSettings.Enabled
            btnFOVCircle.Text = FOVSettings.Enabled and "ON" or "OFF"
            btnFOVCircle.TextColor3 = FOVSettings.Enabled and Color3.fromRGB(0, 255, 128) or Color3.fromRGB(255, 50, 50)
        elseif input.KeyCode == Keybinds.PlayerCam then
            if CamStates.PlayerCamActive then
                CamStates.PlayerCamActive = false
                CamStates.LockedPlayerTarget = nil
            else
                local val = findAimedPlayerTarget()
                if val then
                    CamStates.PlayerCamActive = true
                    CamStates.LockedPlayerTarget = val
                    CamStates.NpcCamActive = false
                    CamStates.LockedNpcTarget = nil
                end
            end
        elseif input.KeyCode == Keybinds.NpcCam then
            if CamStates.NpcCamActive then
                CamStates.NpcCamActive = false
                CamStates.LockedNpcTarget = nil
            else
                local val = findAimedNpcTarget()
                if val then
                    CamStates.NpcCamActive = true
                    CamStates.LockedNpcTarget = val
                    CamStates.PlayerCamActive = false
                    CamStates.LockedPlayerTarget = nil
                end
            end
        elseif input.KeyCode == Keybinds.TpEnemy then
            local closestRoot = nil
            local shortestDist = math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= localPlayer and not isAlly(plr) and plr.Character then
                    local enemyRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                    if enemyRoot and root then
                        local dist = (enemyRoot.Position - root.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestRoot = enemyRoot
                        end
                    end
                end
            end
            if closestRoot and root then
                local behindCFactor = closestRoot.CFrame * CFrame.new(0, 0, 4)
                root.CFrame = CFrame.new(behindCFactor.Position, closestRoot.Position)
            end
        elseif input.KeyCode == Keybinds.SuperJump and root then
            root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, 350, root.AssemblyLinearVelocity.Z)
        elseif input.KeyCode == Keybinds.LongDash and root then
            local lookVec = camera.CFrame.LookVector
            root.AssemblyLinearVelocity = Vector3.new(lookVec.X * 350, 0, lookVec.Z * 350)
        end
    end)
end)
