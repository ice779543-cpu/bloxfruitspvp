local function IsAlive(character)
    return character and character:FindFirstChild("Humanoid") and character.Humanoid.Health > 0
end

-- Function to check if a character or player is using Portal
local function IsUsingPortal(targetModel)
    if not targetModel then return false end
    
    local tool = targetModel:FindFirstChildOfClass("Tool")
    if tool then
        local nameLower = tool.Name:lower()
        if nameLower:find("portal") or nameLower:find("door") then
            return true
        end
    end
    
    local player = Players:GetPlayerFromCharacter(targetModel)
    if player then
        local charFolder = player.Character
        if charFolder then
            for _, item in ipairs(charFolder:GetChildren()) do
                if item:IsA("Tool") and (item.Name:lower():find("portal") or item.Name:lower():find("door")) then
                    return true
                end
            end
        end
    end
    
    return false
end

-- Fast Attack Classes
local FastAttack1 = { Distance = 100 }

function FastAttack1:Attack(BasePart, OthersEnemies)
    if not BasePart or #OthersEnemies == 0 or not RegisterAttack or not RegisterHit then return end
    pcall(function()
        RegisterAttack:FireServer(0)
        RegisterHit:FireServer(BasePart, OthersEnemies)
    end)
end

function FastAttack1:GetNearestEnemies()
    local OthersEnemies = {}
    local EnemiesFolder = Workspace:FindFirstChild("Enemies")
    local CharactersFolder = Workspace:FindFirstChild("Characters")
    local BasePart = nil

    local function ProcessFolder(folder)
        if not folder then return BasePart end
        for _, enemy in ipairs(folder:GetChildren()) do
            local head = enemy:FindFirstChild("Head")
            if head and IsAlive(enemy) and enemy ~= localPlayer.Character and not IsUsingPortal(enemy) then
                if localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    local dist = (localPlayer.Character.HumanoidRootPart.Position - head.Position).Magnitude
                    if dist < FastAttack1.Distance then
                        table.insert(OthersEnemies, { enemy, head })
                        BasePart = head
                    end
                end
            end
        end
        return BasePart
    end

    if EnemiesFolder then ProcessFolder(EnemiesFolder) end
    if CharactersFolder then ProcessFolder(CharactersFolder) end

    return OthersEnemies, BasePart
end

function FastAttack2:AttackNearest()
    local OthersEnemies = {}
    local EnemiesFolder = Workspace:FindFirstChild("Enemies")
    local CharactersFolder = Workspace:FindFirstChild("Characters")
    local BasePart = nil

    local function ProcessFolder(folder)
        if not folder then return nil end
        for _, enemy in ipairs(folder:GetChildren()) do
            local head = enemy:FindFirstChild("Head")
            if head and IsAlive(enemy) and enemy ~= localPlayer.Character and not IsUsingPortal(enemy) then
                if localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    local dist = (localPlayer.Character.HumanoidRootPart.Position - head.Position).Magnitude
                    if dist < FastAttack2.Distance then
                        table.insert(OthersEnemies, { enemy, head })
                        BasePart = head
                    end
                end
            end
        end
        return BasePart
    end

    local Part1 = ProcessFolder(EnemiesFolder)
    local Part2 = ProcessFolder(CharactersFolder)
    
    if #OthersEnemies > 0 then
        self:Attack(Part1 or Part2, OthersEnemies)
    end
end

local function StartAttacksWithMode(mode)
    StopAllAttacks()
    
    local delayTime = AttackDelays[mode] or 0.15
    _G.Settings.FastAttackMode = mode
    
    _G.FastAttackLoop1 = task.spawn(function()
        while _G.Settings.FastAttack do
            local char = localPlayer.Character
            if char and IsAlive(char) then
                local tool = char:FindFirstChildOfClass("Tool")
                if tool and tool.ToolTip ~= "Gun" then
                    local enemies, basePart = FastAttack1:GetNearestEnemies()
                    if #enemies > 0 and basePart then
                        FastAttack1:Attack(basePart, enemies)
                    end
                end
            end
            task.wait(delayTime)
        end
    end)
    
    _G.FastAttackLoop2 = task.spawn(function()
        while _G.Settings.FastAttack do
            local char = localPlayer.Character
            if char and IsAlive(char) then
                local tool = char:FindFirstChildOfClass("Tool")
                if tool and tool.ToolTip ~= "Gun" then
                    FastAttack2:AttackNearest()
                end
            end
            task.wait(delayTime)
        end
    end)
    
    _G.AutoClickLoop = task.spawn(function()
        while _G.Settings.AutoClick do
            local char = localPlayer.Character
            if char and IsAlive(char) and char:FindFirstChildOfClass("Tool") then
                pcall(function()
                    VirtualUser:CaptureController()
                    VirtualUser:Button1Down(Vector2.new(1280, 672))
                end)
            end
            task.wait(0.1)
        end
    end)
end

-- Integrated Custom PreventStun System
local function PreventStun()
    if not NoStunSettings.Enabled then return end
    
    local Character = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local Humanoid = Character:FindFirstChild("Humanoid")
    if Humanoid then
        Humanoid.StateChanged:Connect(function(_, newState)
            if NoStunSettings.Enabled and (newState == Enum.HumanoidStateType.FallingDown or newState == Enum.HumanoidStateType.Ragdoll) then
                Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end)
    end
end

localPlayer.CharacterAdded:Connect(PreventStun)
if localPlayer.Character then
    PreventStun()
end

-- Save / Load Functions
local function saveKeybinds()
    if writefile and HttpService then
        local encodedTable = {}
        for keyName, keyCode in pairs(Keybinds) do encodedTable[keyName] = keyCode.Name end
        pcall(function() writefile(saveFileName, HttpService:JSONEncode(encodedTable)) end)
    end
end

local function loadKeybinds()
    if writefile and readfile and isfile and isfile(saveFileName) then
        local success, decoded = pcall(function() return HttpService:JSONDecode(readfile(saveFileName)) end)
        if success and type(decoded) == "table" then
            for keyName, enumName in pairs(decoded) do
                if Keybinds[keyName] ~= nil and Enum.KeyCode[enumName] then Keybinds[keyName] = Enum.KeyCode[enumName] end
            end
        end
    end
end
loadKeybinds()

-- Dynamic Anti-Ally Checker
local function isPlayerAlly(player)
    if player == localPlayer then return true end

    local success, isAlly = pcall(function()
        if player:GetAttribute("CrewMember") == true then return true end
        if player:GetAttribute("Ally") == true then return true end
        return false
    end)
    
    if success and isAlly then return true end

    if localPlayer.Team and player.Team and localPlayer.Team == player.Team then
        if localPlayer.Team.Name:lower() ~= "pirates" then
            return true
        end
    end

    return false
end

-- Billboard ESP System
local activeESPs = {}

local function getPlayerColor(player)
    if isPlayerAlly(player) then
        return Color3.fromRGB(0, 255, 128)
    end
    local team = player.Team
    if team and team.Name:lower():find("marine") then
        return Color3.fromRGB(0, 150, 255)
    end
    return Color3.fromRGB(255, 50, 50)
end

local function createESP(player)
    if player == localPlayer then return end

    local function setupCharacterESP(char)
        if activeESPs[player] then
            if activeESPs[player].billboard then activeESPs[player].billboard:Destroy() end
            if activeESPs[player].conn then activeESPs[player].conn:Disconnect() end
            activeESPs[player] = nil
        end

        local adorneePart = char:WaitForChild("Head", 5) or char:WaitForChild("HumanoidRootPart", 5) or char:FindFirstChildOfClass("BasePart")
        local humanoid = char:WaitForChild("Humanoid", 5)
        if not adorneePart or not humanoid then return end

        local billboard = Instance.new("BillboardGui")
        billboard.Name = "PlayerESP"
        billboard.Size = UDim2.new(0, 180, 0, 45)
        billboard.StudsOffset = Vector3.new(0, 2.5, 0)
        billboard.AlwaysOnTop = true
        billboard.Adornee = adorneePart
        billboard.Parent = CoreGui

        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextSize = 13
        textLabel.TextColor3 = getPlayerColor(player)
        textLabel.TextStrokeTransparency = 0
        textLabel.TextXAlignment = Enum.TextXAlignment.Center
        textLabel.TextYAlignment = Enum.TextYAlignment.Center
        textLabel.Parent = billboard

        local elapsed = 0
        local connection
        connection = RunService.Heartbeat:Connect(function(dt)
            elapsed = elapsed + dt
            if elapsed < 0.2 then return end
            elapsed = 0

            if not char or not char.Parent or not humanoid or humanoid.Health <= 0 then
                billboard:Destroy()
                connection:Disconnect()
                activeESPs[player] = nil
                return
            end

            local lvl = "N/A"
            if player:FindFirstChild("leaderstats") and player.leaderstats:FindFirstChild("Level") then
                lvl = player.leaderstats.Level.Value
            elseif player:FindFirstChild("Data") and player.Data:FindFirstChild("Level") then
                lvl = player.Data.Level.Value
            end

            textLabel.TextColor3 = getPlayerColor(player)
            textLabel.Text = string.format("%s [Lvl. %s]\nHP: %d/%d", player.Name, tostring(lvl), math.floor(humanoid.Health), math.floor(humanoid.MaxHealth))
        end)

        activeESPs[player] = {billboard = billboard, conn = connection}
    end

    if player.Character then
        setupCharacterESP(player.Character)
    end
    player.CharacterAdded:Connect(setupCharacterESP)
end

for _, p in ipairs(Players:GetPlayers()) do
    createESP(p)
end
Players.PlayerAdded:Connect(createESP)

Players.PlayerRemoving:Connect(function(player)
    if activeESPs[player] then
        if activeESPs[player].billboard then activeESPs[player].billboard:Destroy() end
        if activeESPs[player].conn then activeESPs[player].conn:Disconnect() end
        activeESPs[player] = nil
    end
end)

local playerCamlockActive, npcCamlockActive, currentTarget = false, false, nil

local function getClosestPlayer()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local myRootPos = char.HumanoidRootPart.Position

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and not isPlayerAlly(p) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (myRootPos - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortest then shortest, closest = dist, p.Character end
            end
        end
    end
    return closest
end

local function getClosestAlly()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local myRootPos = char.HumanoidRootPart.Position

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and isPlayerAlly(p) then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (myRootPos - p.Character.HumanoidRootPart.Position).Magnitude
                if dist < shortest then shortest, closest = dist, p.Character end
            end
        end
    end
    return closest
end

local function getClosestNPC()
    local closest, shortest = nil, math.huge
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local myRootPos = char.HumanoidRootPart.Position

    local enemiesFolderTarget = Workspace:FindFirstChild("Enemies") or Workspace
    for _, obj in ipairs(enemiesFolderTarget:GetChildren()) do
        if obj:IsA("Model") and obj ~= char then
            local hum, root = obj:FindFirstChildOfClass("Humanoid"), obj:FindFirstChild("HumanoidRootPart")
            if hum and root and not Players:GetPlayerFromCharacter(obj) and hum.Health > 0 then
                local dist = (myRootPos - root.Position).Magnitude
                if dist < shortest then shortest, closest = dist, obj end
            end
        end
    end
    return closest
end

local function updateStatusText()
    if playerCamlockActive then StatusLabel.Text, StatusLabel.TextColor3 = "Cam: Player ON", Color3.fromRGB(0, 255, 128)
    elseif npcCamlockActive then StatusLabel.Text, StatusLabel.TextColor3 = "Cam: NPC ON", Color3.fromRGB(0, 255, 128)
    else StatusLabel.Text, StatusLabel.TextColor3 = "Cam: OFF", Color3.fromRGB(255, 50, 50) end
end

UserInputService.InputBegan:Connect(function(input, gp)
    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
    if UserInputService:GetFocusedTextBox() then return end

    if listeningKey then
        if input.KeyCode ~= Enum.KeyCode.Escape then
            Keybinds[listeningKey] = input.KeyCode
            buttonMap[listeningKey].Text = input.KeyCode.Name
            saveKeybinds()
        else
            buttonMap[listeningKey].Text = Keybinds[listeningKey].Name
        end
        buttonMap[listeningKey].TextColor3 = Color3.fromRGB(0, 255, 128)
        listeningKey = nil
        return
    end

    if input.KeyCode == Keybinds.MenuToggle then toggleMenu() return end
    if input.KeyCode == Keybinds.HideToggle then
        ScreenGui.Enabled = not ScreenGui.Enabled
        return
    end
    if input.KeyCode == Keybinds.NoStunKey then
        toggleNoStun()
        return
    end
    
    if gp then return end

    if input.KeyCode == Keybinds.PlayerCam then
        npcCamlockActive = false
        playerCamlockActive = not playerCamlockActive
        currentTarget = playerCamlockActive and getClosestPlayer() or nil
        if playerCamlockActive and not currentTarget then playerCamlockActive = false end
        updateStatusText()
    elseif input.KeyCode == Keybinds.NpcCam then
        playerCamlockActive = false
        npcCamlockActive = not npcCamlockActive
        currentTarget = npcCamlockActive and getClosestNPC() or nil
        if npcCamlockActive and not currentTarget then npcCamlockActive = false end
        updateStatusText()
    elseif input.KeyCode == Keybinds.SuperJump then
        local char = localPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(char.HumanoidRootPart.AssemblyLinearVelocity.X, 180, char.HumanoidRootPart.AssemblyLinearVelocity.Z)
        end
    elseif input.KeyCode == Keybinds.LongDash then
        local char = localPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local look = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z).Unit
            char.HumanoidRootPart.AssemblyLinearVelocity = look * 350 + Vector3.new(0, 20, 0)
        end
    elseif input.KeyCode == Keybinds.TpEnemy then
        local tChar = getClosestPlayer()
        if localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart") and tChar and tChar:FindFirstChild("HumanoidRootPart") then
            localPlayer.Character.HumanoidRootPart.CFrame = tChar.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
        end
    elseif input.KeyCode == Keybinds.TpAlly then
        local tChar = getClosestAlly()
        if localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart") and tChar and tChar:FindFirstChild("HumanoidRootPart") then
            localPlayer.Character.HumanoidRootPart.CFrame = tChar.HumanoidRootPart.CFrame * CFrame.new(0, 0, -3)
        end
    end
end)

RunService.RenderStepped:Connect(function(dt)
    if (playerCamlockActive or npcCamlockActive) and currentTarget and currentTarget:FindFirstChild("HumanoidRootPart") then
        local humanoid = currentTarget:FindFirstChildOfClass("Humanoid")
        local targetPlayer = Players:GetPlayerFromCharacter(currentTarget)
        
        if not humanoid or humanoid.Health <= 0 or (playerCamlockActive and targetPlayer and isPlayerAlly(targetPlayer)) then
            playerCamlockActive = false
            npcCamlockActive = false
            currentTarget = nil
            updateStatusText()
            return
        end

        local targetRoot = currentTarget.HumanoidRootPart
        local currentCamCFrame = camera.CFrame
        camera.CFrame = currentCamCFrame:Lerp(CFrame.new(currentCamCFrame.Position, targetRoot.Position + Vector3.new(0, 0.2, 0)), 1 - math.exp(-35 * dt))
    end
end)
