-- Dynamic and Safe Remote Finder
local function getRegisterAttack()
    local success, res = pcall(function()
        return ReplicatedStorage.Modules.Net["RE/RegisterAttack"]
    end)
    if success and res then return res end
    
    for _, v in ipairs(ReplicatedStorage:GetDescendants()) do
        if v.Name == "RegisterAttack" or v.Name == "RE/RegisterAttack" then
            return v
        end
    end
    return nil
end

local function getRegisterHit()
    local success, res = pcall(function()
        return ReplicatedStorage.Modules.Net["RE/RegisterHit"]
    end)
    if success and res then return res end
    
    for _, v in ipairs(ReplicatedStorage:GetDescendants()) do
        if v.Name == "RegisterHit" or v.Name == "RE/RegisterHit" then
            return v
        end
    end
    return nil
end

local function executeAttack(targetRoot, targetHead)
    local char = localPlayer.Character
    if not char then return end
    
    local equippedWeapon = char:FindFirstChildOfClass("Tool")
    if equippedWeapon then
        pcall(function()
            local attackRemote = equippedWeapon:FindFirstChild("LeftClickRemote") 
                or equippedWeapon:FindFirstChild("ToolAttack") 
                or equippedWeapon:FindFirstChild("AttackRemote")
                
            if attackRemote and typeof(attackRemote.FireServer) == "function" and targetRoot then
                local direction = (targetRoot.Position - char:GetPivot().Position).Unit
                attackRemote:FireServer(direction, 1)
            end
        end)
    end
    
    pcall(function()
        local regAttack = getRegisterAttack()
        if regAttack and typeof(regAttack.FireServer) == "function" then
            regAttack:FireServer(0)
        end
        
        local regHit = getRegisterHit()
        if regHit and typeof(regHit.FireServer) == "function" and targetHead then
            regHit:FireServer(targetHead, { {targetHead.Parent, targetHead} })
        end
    end)
end

local function saveKeybinds()
    if writefile and HttpService then
        local encodedTable = {}
        for keyName, keyCode in pairs(Keybinds) do encodedTable[keyName] = keyCode.Name end
        pcall(function() writefile(saveFileName, HttpService:JSONEncode(encodedTable)) end)
    end
end

local function loadKeybinds()
    if writefile and readfile and isfile and isfile(saveFileName) then
        local success, decoded = pcall(function() return HttpService:JSONDecode(readfile(saveFileName)) end)
        if success and type(decoded) == "table" then
            for keyName, enumName in pairs(decoded) do
                if Keybinds[keyName] ~= nil and Enum.KeyCode[enumName] then Keybinds[keyName] = Enum.KeyCode[enumName] end
            end
        end
    end
end

-- THROTTLED FAST ATTACK LOOP
task.spawn(function()
    while true do
        if FastAttackSettings.AutoClick then
            local char = localPlayer.Character
            if char and char:FindFirstChildOfClass("Tool") then
                local myRoot = char:FindFirstChild("HumanoidRootPart")
                
                for _, plr in ipairs(Players:GetPlayers()) do
                    if plr ~= localPlayer and plr.Character then
                        local root = plr.Character:FindFirstChild("HumanoidRootPart")
                        local head = plr.Character:FindFirstChild("Head")
                        local hum = plr.Character:FindFirstChild("Humanoid")
                        if root and head and hum and hum.Health > 0 and myRoot and (root.Position - myRoot.Position).Magnitude <= FastAttackSettings.Distance then
                            executeAttack(root, head)
                        end
                    end
                end

                local enemiesFolder = Workspace:FindFirstChild("Enemies")
                if enemiesFolder then
                    for _, mob in pairs(enemiesFolder:GetChildren()) do
                        local head = mob:FindFirstChild("Head")
                        local root = mob:FindFirstChild("HumanoidRootPart")
                        local hum = mob:FindFirstChild("Humanoid")
                        if head and root and hum and hum.Health > 0 and myRoot and (root.Position - myRoot.Position).Magnitude <= FastAttackSettings.Distance then
                            executeAttack(root, head)
                        end
                    end
                end
            end
        end
        task.wait(FastAttackSettings.ClickDelay)
    end
end)

-- Raid Automation Script
local currentTween = nil
local function stopTween()
    if currentTween then
        currentTween:Cancel()
        currentTween = nil
    end
end

local function smoothFlyTo(targetCFrame, speed)
    local char = localPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    stopTween()
    local distance = (root.Position - targetCFrame.Position).Magnitude
    local timeTaken = distance / (speed or 250)
    
    currentTween = TweenService:Create(root, TweenInfo.new(timeTaken, Enum.EasingStyle.Linear), {CFrame = targetCFrame})
    currentTween:Play()
end

task.spawn(function()
    while true do
        task.wait(0.2)
        pcall(function()
            if RaidStates.AutoFarmRaid then
                local playerGui = localPlayer:FindFirstChild("PlayerGui")
                local raidTimer = playerGui and playerGui:FindFirstChild("Main") and playerGui.Main:FindFirstChild("TopHUDList") and playerGui.Main.TopHUDList:FindFirstChild("RaidTimer")
                
                if raidTimer and raidTimer.Visible == false then
                    local function GetBP(itemName)
                        for _, v in ipairs(localPlayer.Backpack:GetChildren()) do
                            if v.Name == itemName then return v end
                        end
                        if localPlayer.Character and localPlayer.Character:FindFirstChild(itemName) then
                            return localPlayer.Character[itemName]
                        end
                        return nil
                    end

                    if GetBP("Special Microchip") then
                        local World2 = Workspace:FindFirstChild("Map") and Workspace.Map:FindFirstChild("CircleIsland")
                        if World2 then
                            smoothFlyTo(CFrame.new(-6438.73535, 250.645355, -4501.50684), 300)
                            task.wait(1)
                            local btn = Workspace.Map.CircleIsland:FindFirstChild("RaidSummon2")
                            if btn and btn:FindFirstChild("Button") and btn.Button:FindFirstChild("Main") and btn.Button.Main:FindFirstChild("ClickDetector") then
                                fireclickdetector(btn.Button.Main.ClickDetector)
                            end
                        else
                            pcall(function()
                                ReplicatedStorage.Remotes.CommF_:InvokeServer("requestEntrance", Vector3.new(-5097.93164, 316.447021, -3142.66602))
                            end)
                            task.wait(0.5)
                            smoothFlyTo(CFrame.new(-5033.50879, 315.014252, -2947.77539), 300)
                            local boatCastle = Workspace:FindFirstChild("Map") and Workspace.Map:FindFirstChild("Boat Castle")
                            if boatCastle and boatCastle:FindFirstChild("RaidSummon2") then
                                local clickDet = boatCastle.RaidSummon2:FindFirstChild("Button") and boatCastle.RaidSummon2.Button:FindFirstChild("Main") and boatCastle.RaidSummon2.Button.Main:FindFirstChild("ClickDetector")
                                if clickDet then fireclickdetector(clickDet) end
                            end
                        end
                    end
                end

                if raidTimer and raidTimer.Visible == true then
                    local enemiesFolder = Workspace:FindFirstChild("Enemies")
                    local targetMobRoot, targetMobHead = nil, nil
                    
                    if enemiesFolder then
                        for _, mob in pairs(enemiesFolder:GetChildren()) do
                            local hum = mob:FindFirstChild("Humanoid")
                            local root = mob:FindFirstChild("HumanoidRootPart")
                            local head = mob:FindFirstChild("Head")
                            if hum and root and head and hum.Health > 0 then
                                targetMobRoot = root
                                targetMobHead = head
                                break
                            end
                        end
                    end

                    local character = localPlayer.Character
                    local myRoot = character and character:FindFirstChild("HumanoidRootPart")

                    if targetMobRoot and targetMobHead and myRoot then
                        smoothFlyTo(targetMobRoot.CFrame * CFrame.new(0, 12, 0), 280)
                        executeAttack(targetMobRoot, targetMobHead)
                    else
                        stopTween()
                        if myRoot then myRoot.Anchored = false end
                    end
                else
                    stopTween()
                    local character = localPlayer.Character
                    local myRoot = character and character:FindFirstChild("HumanoidRootPart")
                    if myRoot then myRoot.Anchored = false end
                end
            end
        end)
    end
end)

-- Fixed Keybinds & Camlock/Dash Logic
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    local char = localPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if input.KeyCode == Keybinds.MenuToggle then
        MainFrame.Visible = not MainFrame.Visible
    elseif input.KeyCode == Keybinds.MenuHide then
        ScreenGui.Enabled = not ScreenGui.Enabled
    elseif input.KeyCode == Keybinds.NoStunKey then
        NoStunSettings.Enabled = not NoStunSettings.Enabled
        btnNoStun.Text = NoStunSettings.Enabled and "ON" or "OFF"
        btnNoStun.TextColor3 = NoStunSettings.Enabled and Color3.fromRGB(0, 255, 128) or Color3.fromRGB(255, 50, 50)
    elseif input.KeyCode == Keybinds.SuperJump and root then
        root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, 350, root.AssemblyLinearVelocity.Z)
    elseif input.KeyCode == Keybinds.LongDash and root then
        -- Repaired Long Dash implementation utilizing updated AssemblyLinearVelocity physics safely
        local lookVec = camera.CFrame.LookVector
        root.AssemblyLinearVelocity = Vector3.new(lookVec.X * 350, root.AssemblyLinearVelocity.Y, lookVec.Z * 350)
    elseif input.KeyCode == Keybinds.PlayerCam then
        -- Repaired Player Camlock (Toggles locking onto the closest player)
        if PvPStates.CamlockTarget then
            PvPStates.CamlockTarget = nil
        else
            local closestPlayer = nil
            local shortestDist = math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= localPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                    local dist = (plr.Character.HumanoidRootPart.Position - root.Position).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        closestPlayer = plr
                    end
                end
            end
            PvPStates.CamlockTarget = closestPlayer
        end
    elseif input.KeyCode == Keybinds.TpEnemy and root then
        local closestEnemy = nil
        local shortestDistance = math.huge
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= localPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                local dist = (plr.Character.HumanoidRootPart.Position - root.Position).Magnitude
                if dist < shortestDistance then
                    shortestDistance = dist
                    closestEnemy = plr.Character.HumanoidRootPart
                end
            end
        end
        if closestEnemy then
            root.CFrame = closestEnemy.CFrame * CFrame.new(0, 0, 3)
        end
    end
end)

    MainFrame.Visible = not MainFrame.Visible
end)
