local function isValidRaidEnemy(mob)
    local success, result = pcall(function()
        if not mob or not mob.Parent then return false end
        local hum = mob:FindFirstChild("Humanoid")
        local root = mob:FindFirstChild("HumanoidRootPart")
        if not hum or not root then return false end
        if hum.Health <= 0 then return false end
        return true
    end)
    return success and result
end

local function getRegisterAttack()
    local success, res = pcall(function() return ReplicatedStorage.Modules.Net["RE/RegisterAttack"] end)
    if success and res then return res end
    for _, v in ipairs(ReplicatedStorage:GetDescendants()) do
        if v.Name == "RegisterAttack" or v.Name == "RE/RegisterAttack" then return v end
    end
    return nil
end

end

local function executeMultiAttack(targetsList)
    pcall(function()
        local char = localPlayer.Character
        if not char then return end
        local equippedWeapon = char:FindFirstChildOfClass("Tool")
        if equippedWeapon then
            local attackRemote = equippedWeapon:FindFirstChild("LeftClickRemote") or equippedWeapon:FindFirstChild("ToolAttack") or equippedWeapon:FindFirstChild("AttackRemote")
            if attackRemote and typeof(attackRemote.FireServer) == "function" and #targetsList > 0 then
                local firstRoot = targetsList[1].Root
                local direction = (firstRoot.Position - char:GetPivot().Position).Unit
                attackRemote:FireServer(direction, 1)
            end
        end
        
        local regAttack = getRegisterAttack()
        if regAttack and typeof(regAttack.FireServer) == "function" then regAttack:FireServer(0) end
        
        local regHit = getRegisterHit()
        if regHit and typeof(regHit.FireServer) == "function" and #targetsList > 0 then
            local hitParams = {}
            for _, data in ipairs(targetsList) do
                if data.Head and data.Head.Parent then
                    table.insert(hitParams, {data.Head.Parent, data.Head})
                end
            end
            if #hitParams > 0 then
                regHit:FireServer(targetsList[1].Head, hitParams)
            end
        end
    end)
end

local function equipToolType(toolType)
    local char = localPlayer.Character
    if not char then return end

    local currentTool = char:FindFirstChildOfClass("Tool")
    if currentTool then
        local tip = currentTool.ToolTip:lower()
        if tip == toolType.lower() or (toolType == "Fruit" and tip:find("blox fruit")) then
            return
        end
    end

    local foundTool = nil
    local containerList = {localPlayer.Backpack, char}

    for _, container in ipairs(containerList) do
        for _, t in ipairs(container:GetChildren()) do
            if t:IsA("Tool") then
                local tip = t.ToolTip:lower()
                local name = t.Name:lower()
                
                if toolType == "Sword" and (tip == "sword" or name:find("sword")) then
                    foundTool = t; break
                elseif toolType == "Fruit" and (tip:find("blox fruit") or tip == "fruit" or name:find("fruit") or name:find("portal") or name:find("dough") or name:find("kitsune") or name:find("leopard") or name:find("trex")) then
                    foundTool = t; break
                elseif toolType == "Gun" and (tip == "gun" or name:find("gun") or name:find("rifle") or name:find("musket") or name:find("bazooka")) then
                    foundTool = t; break
                elseif toolType == "Melee" and (tip == "blox fighting style" or tip == "melee" or name:find("fighting") or name:find("step") or name:find("claw")) then
                    foundTool = t; break
                end
            end
        end
        if foundTool then break end
    end

    if not foundTool and toolType == "Melee" then
        for _, container in ipairs(containerList) do
            for _, t in ipairs(container:GetChildren()) do
                if t:IsA("Tool") and (t.ToolTip == "" or t.ToolTip:lower() == "melee") then
                    foundTool = t; break
                end
            end
            if foundTool then break end
        end
    end

    if foundTool then
        if foundTool.Parent == localPlayer.Backpack then
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid then
                humanoid:EquipTool(foundTool)
                task.wait(0.25)
            end
        end
    end
end

local function runMacroSequence()
    task.spawn(function()
        for _, actionStr in ipairs(MacroSelections) do
            if actionStr and actionStr ~= "None" then
                local spacePos = actionStr:find(" ")
                if spacePos then
                    local category = actionStr:sub(1, spacePos - 1)
                    local moveKey = actionStr:sub(spacePos + 1):upper()

                    equipToolType(category)
                    task.wait(0.1)

                    local keyCode = Enum.KeyCode[moveKey]
                    if keyCode then
                        pcall(function()
                            VirtualInputManager:SendKeyEvent(true, keyCode, false, game)
                            task.wait(0.05)
                            VirtualInputManager:SendKeyEvent(false, keyCode, false, game)
                        end)
                        -- Increased wait interval between actions so consecutive moves don't cancel/fail
                        task.wait(0.35)
                    end
                end
            end
        end
    end)
end

local function findClosestPlayerTarget()
    local closestRoot = nil
    local shortestDist = math.huge
    local myRoot = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= localPlayer and plr.Character then
            local enemyRoot = plr.Character:FindFirstChild("HumanoidRootPart")
            local hum = plr.Character:FindFirstChild("Humanoid")
            if enemyRoot and hum and hum.Health > 0 then
                local dist = (enemyRoot.Position - myRoot.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    closestRoot = enemyRoot
                end
            end
        end
    end
    return closestRoot
end

local function findClosestNpcTarget()
    local closestRoot = nil
    local shortestDist = math.huge
    local myRoot = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end

    local enemiesFolder = Workspace:FindFirstChild("Enemies")
    if enemiesFolder then
        for _, mob in ipairs(enemiesFolder:GetChildren()) do
            if isValidRaidEnemy(mob) then
                local enemyRoot = mob:FindFirstChild("HumanoidRootPart")
                if enemyRoot then
                    local dist = (enemyRoot.Position - myRoot.Position).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        closestRoot = enemyRoot
                    end
                end
            end
        end
    end
    return closestRoot
end

local function findCurrentRaidIsland(rootPos)
    local bestTarget = nil
    local shortestDist = math.huge

    local enemiesFolder = Workspace:FindFirstChild("Enemies")
    if enemiesFolder then
        for _, mob in ipairs(enemiesFolder:GetChildren()) do
            if isValidRaidEnemy(mob) then
                local mobRoot = mob:FindFirstChild("HumanoidRootPart")
                if mobRoot then
                    local dist = (mobRoot.Position - rootPos).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        bestTarget = mobRoot.Position + Vector3.new(0, 20, 0)
                    end
                end
            end
        end
    end

    if bestTarget then return bestTarget end

    for _, obj in ipairs(Workspace:GetChildren()) do
        local name = obj.Name:lower()
        if name:find("island") or name:find("raid") or name:find("map") then
            if obj:IsA("Model") then
                local success, pivot = pcall(function() return obj:GetPivot().Position end)
                if success and pivot then
                    local dist = (pivot - rootPos).Magnitude
                    if dist < shortestDist and pivot.Y > -50 and pivot.Y < 3000 then
                        shortestDist = dist
                        bestTarget = pivot + Vector3.new(0, 40, 0)
                    end
                end
            end
        end
    end

    return bestTarget
end

local function manageRaidFlight()
    pcall(function()
        local char = localPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChild("Humanoid")
        if not root or not hum or hum.Health <= 0 then return end

        local bv = root:FindFirstChild("RaidFlyBV")
        local bg = root:FindFirstChild("RaidFlyBG")

        if AutoRaidSettings.Enabled then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end

            if not bv then
                bv = Instance.new("BodyVelocity")
                bv.Name = "RaidFlyBV"
                bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                bv.Velocity = Vector3.new(0, 0, 0)
                bv.Parent = root
            end
            if not bg then
                bg = Instance.new("BodyGyro")
                bg.Name = "RaidFlyBG"
                bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                bg.CFrame = root.CFrame
                bg.Parent = root
            end

            local targetPos = findCurrentRaidIsland(root.Position)

            if targetPos then
                local direction = (targetPos - root.Position)
                if direction.Magnitude > 10 then
                    bv.Velocity = direction.Unit * AutoRaidSettings.FlySpeed
                    bg.CFrame = CFrame.new(root.Position, targetPos)
                else
                    bv.Velocity = Vector3.new(0, 0, 0)
                end
            else
                bv.Velocity = Vector3.new(0, 0, 0)
            end
        else
            if bv then bv:Destroy() end
            if bg then bg:Destroy() end
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = true end
            end
        end
    end)
end

local function saveSettings()
    pcall(function()
        if writefile and HttpService then
            local encodedTable = {
                Keybinds = {},
                MacroSelections = MacroSelections
            }
            for keyName, keyCode in pairs(Keybinds) do 
                encodedTable.Keybinds[keyName] = keyCode.Name 
            end
            writefile(saveFileName, HttpService:JSONEncode(encodedTable))
        end
    end)
end

local function loadSettings()
    pcall(function()
        if readfile and isfile and isfile(saveFileName) then
            local success, decoded = pcall(function() return HttpService:JSONDecode(readfile(saveFileName)) end)
            if success and type(decoded) == "table" then
                if decoded.Keybinds then
                    for keyName, enumName in pairs(decoded.Keybinds) do
                        if Keybinds[keyName] ~= nil and Enum.KeyCode[enumName] then 
                            Keybinds[keyName] = Enum.KeyCode[enumName] 
                        end
                    end
                end
                if decoded.MacroSelections and type(decoded.MacroSelections) == "table" then
                    for i = 1, #MacroSelections do
                        if decoded.MacroSelections[i] then
                            MacroSelections[i] = decoded.MacroSelections[i]
                        end
                    end
                end
            end
        end
    end)
end

    Fill.Size = UDim2.new((currentVal - minVal) / (maxVal - minVal), 0, 1, 0)

    Fill.BorderSizePixel = 0
    Fill.Parent = SliderBar
    Instance.new("UICorner", Fill).CornerRadius = UDim.new(0, 4)

    local dragging = false
    SliderBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = true end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local mousePos = UserInputService:GetMouseLocation()
            local relX = math.clamp((mousePos.X - SliderBar.AbsolutePosition.X) / SliderBar.AbsoluteSize.X, 0, 1)
            local val = math.floor(minVal + ((maxVal - minVal) * relX))
            Fill.Size = UDim2.new(relX, 0, 1, 0)
            Label.Text = name .. ": " .. val
            callback(val)
        end
    end)

local function getTargetInsideFOV()
    local bestTarget = nil
    local shortestDist = math.huge
    local screenSize = camera.ViewportSize
    local screenCenter = Vector2.new(screenSize.X / 2, screenSize.Y / 2)

    FOVCircle.Position = screenCenter
    FOVCircle.Visible = FOVSettings.Enabled and MainFrame.Visible == false

    if not FOVSettings.Enabled then return nil end

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= localPlayer and plr.Character then
            local root = plr.Character:FindFirstChild("HumanoidRootPart")
            local hum = plr.Character:FindFirstChild("Humanoid")
            if root and hum and hum.Health > 0 then
                local screenPos, onScreen = camera:WorldToViewportPoint(root.Position)
                if onScreen then
                    local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                    if screenDist <= FOVSettings.Radius and screenDist < shortestDist then
                        shortestDist = screenDist
                        bestTarget = root
                    end
                end
            end
        end
    end

    return bestTarget
end

task.spawn(function()
    while true do
        pcall(function()
            if FastAttackSettings.AutoClick then
                local char = localPlayer.Character
                if char and char:FindFirstChildOfClass("Tool") then
                    local myRoot = char:FindFirstChild("HumanoidRootPart")
                    if myRoot then
                        local targetsList = {}
                        
                        local enemiesFolder = Workspace:FindFirstChild("Enemies")
                        if enemiesFolder then
                            for _, mob in pairs(enemiesFolder:GetChildren()) do
                                if isValidRaidEnemy(mob) then
                                    local head = mob:FindFirstChild("Head")
                                    local root = mob:FindFirstChild("HumanoidRootPart")
                                    local hum = mob:FindFirstChild("Humanoid")
                                    if head and root and hum and hum.Health > 0 and (root.Position - myRoot.Position).Magnitude <= FastAttackSettings.Distance then
                                        table.insert(targetsList, {Head = head, Root = root})
                                    end
                                end
                            end
                        end

                        for _, plr in ipairs(Players:GetPlayers()) do
                            if plr ~= localPlayer and plr.Character then
                                local head = plr.Character:FindFirstChild("Head")
                                local root = plr.Character:FindFirstChild("HumanoidRootPart")
                                local hum = plr.Character:FindFirstChild("Humanoid")
                                if head and root and hum and hum.Health > 0 and (root.Position - myRoot.Position).Magnitude <= FastAttackSettings.Distance then
                                    table.insert(targetsList, {Head = head, Root = root})
                                end
                            end
                        end

                        if #targetsList > 0 then
                            executeMultiAttack(targetsList)
                        end
                    end
                end
            end
        end)
        task.wait(FastAttackSettings.ClickDelay)
    end
end)

RunService.RenderStepped:Connect(function()
    pcall(function()
        if AutoRaidSettings.Enabled then
            manageRaidFlight()
        end

        -- Handle Player Camlock & NPC Camlock execution loops
        if CamStates.PlayerCamActive then
            local targetRoot = findClosestPlayerTarget()
            if targetRoot then
                camera.CFrame = CFrame.new(camera.CFrame.Position, targetRoot.Position)
            end
        elseif CamStates.NpcCamActive then
            local targetRoot = findClosestNpcTarget()
            if targetRoot then
                camera.CFrame = CFrame.new(camera.CFrame.Position, targetRoot.Position)
            end
        end

        if ESPStates.Enabled then
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= localPlayer then
                    local char = plr.Character
                    local root = char and char:FindFirstChild("HumanoidRootPart")
                    local hum = char and char:FindFirstChild("Humanoid")
                    local existingEsp = espFolder:FindFirstChild("ESP_" .. plr.Name)
                    
                    if char and root and hum and hum.Health > 0 then
                        local playerLevel = "1"
                        local stats = plr:FindFirstChild("Data") or plr:FindFirstChild("leaderstats")
                        if stats and stats:FindFirstChild("Level") then playerLevel = tostring(stats.Level.Value) end

                        local espColor = Color3.fromRGB(255, 50, 50)
                        if plr.Team then
                            local teamName = tostring(plr.Team.Name):lower()
                            if teamName:find("marine") then
                                espColor = Color3.fromRGB(50, 150, 255)
                            end
                        end

                        local billboard = existingEsp
                        if not billboard then
                            billboard = Instance.new("BillboardGui")
                            billboard.Name = "ESP_" .. plr.Name
                            billboard.Size = UDim2.new(0, 150, 0, 40)
                            billboard.StudsOffset = Vector3.new(0, 3, 0)
                            billboard.AlwaysOnTop = true
                            billboard.Parent = espFolder
                            
                            local textLbl = Instance.new("TextLabel")
                            textLbl.Name = "TextInfo"
                            textLbl.Size = UDim2.new(1, 0, 1, 0)
                            textLbl.BackgroundTransparency = 1
                            textLbl.Font = Enum.Font.SourceSansBold
                            textLbl.TextSize = 13
                            textLbl.TextStrokeTransparency = 0.2
                            textLbl.Parent = billboard
                        end

                        billboard.Adornee = root
                        local textLbl = billboard:FindFirstChild("TextInfo")
                        if textLbl then
                            textLbl.TextColor3 = espColor
                            textLbl.Text = string.format("%s [Lvl: %s]\n[%d / %d HP]", plr.Name, playerLevel, math.floor(hum.Health), math.floor(hum.MaxHealth))
                        end
                    else
                        if existingEsp then existingEsp:Destroy() end
                    end
                end
            end
        else
            for _, v in ipairs(espFolder:GetChildren()) do v:Destroy() end
        end
        
        if FOVSettings.Enabled then
            local hue = (tick() * FOVSettings.RainbowSpeed) % 1
            FOVCircle.Color = Color3.fromHSV(hue, 1, 1)

            local target = getTargetInsideFOV()
            if target then
                camera.CFrame = CFrame.new(camera.CFrame.Position, target.Position)
            end
        else
            FOVCircle.Visible = false
        end
    end)
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    pcall(function()
        local char = localPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")

        if input.KeyCode == Keybinds.MenuToggle then
            MainFrame.Visible = not MainFrame.Visible
        elseif input.KeyCode == Keybinds.MenuHide then
            ScreenGui.Enabled = not ScreenGui.Enabled
        elseif input.KeyCode == Keybinds.MacroKey then
            runMacroSequence()
        elseif input.KeyCode == Keybinds.AimbotKey then
            FOVSettings.Enabled = not FOVSettings.Enabled
            btnFOVCircle.Text = FOVSettings.Enabled and "ON" or "OFF"
            btnFOVCircle.TextColor3 = FOVSettings.Enabled and Color3.fromRGB(0, 255, 128) or Color3.fromRGB(255, 50, 50)
        elseif input.KeyCode == Keybinds.PlayerCam then
            CamStates.PlayerCamActive = not CamStates.PlayerCamActive
            if CamStates.PlayerCamActive then CamStates.NpcCamActive = false end
        elseif input.KeyCode == Keybinds.NpcCam then
            CamStates.NpcCamActive = not CamStates.NpcCamActive
            if CamStates.NpcCamActive then CamStates.PlayerCamActive = false end
        elseif input.KeyCode == Keybinds.TpEnemy then
            local closestRoot = nil
            local shortestDist = math.huge
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= localPlayer and plr.Character then
                    local enemyRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                    if enemyRoot and root then
                        local dist = (enemyRoot.Position - root.Position).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestRoot = enemyRoot
                        end
                    end
                end
            end
            if closestRoot and root then
                local behindCFrame = closestRoot.CFrame * CFrame.new(0, 0, 4)
                root.CFrame = CFrame.new(behindCFrame.Position, closestRoot.Position)
            end
        elseif input.KeyCode == Keybinds.SuperJump and root then
            root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, 350, root.AssemblyLinearVelocity.Z)
        elseif input.KeyCode == Keybinds.LongDash and root then
            local lookVec = camera.CFrame.LookVector
            root.AssemblyLinearVelocity = Vector3.new(lookVec.X * 350, 0, lookVec.Z * 350)
        end
    end)
end)
